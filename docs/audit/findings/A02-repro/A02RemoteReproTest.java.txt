// A02 감사 재현(커밋하지 않음): RCON 인증 전 30초 제한이 "마지막 읽기 이후 30초"라 인증 없이 자리를 무기한 붙잡을 수 있는지,
// 그리고 Query 챌린지 상한(1024)을 한 주소의 포트만 바꿔 채우면 정상 질의가 30초 동안 막히는지.
package org.domi.furnace.server.remote;

import static org.junit.jupiter.api.Assertions.*;

import java.io.DataInputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.OutputStream;
import java.net.InetAddress;
import java.net.InetSocketAddress;
import java.net.ServerSocket;
import java.net.Socket;
import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.atomic.AtomicLong;
import org.junit.jupiter.api.Test;

class A02RemoteReproTest {

    private static void send(OutputStream out, int id, int type, String payload) throws IOException {
        out.write(RconProtocol.encode(id, type, payload));
        out.flush();
    }

    private static void drain(InputStream in) throws IOException {
        DataInputStream d = new DataInputStream(in);
        int len = Integer.reverseBytes(d.readInt());
        d.readFully(new byte[len]);
    }

    /** 인증하지 않은 연결이 20초마다 알 수 없는 종류(9) 패킷을 보내면 AUTH_TIMEOUT(30초)이 지나도 끊기지 않는다. */
    @Test
    void unauthenticatedClientOutlivesAuthTimeout() throws Exception {
        ServerSocket ss = new ServerSocket(0, 0, InetAddress.getLoopbackAddress());
        RconServer server = RconServer.open(ss, "secret", cmd -> "ran:" + cmd);
        try (Socket s = new Socket(InetAddress.getLoopbackAddress(), server.port())) {
            s.setSoTimeout(5000);
            long t0 = System.nanoTime();
            for (int i = 0; i < 3; i++) {             // 0초·20초·40초
                send(s.getOutputStream(), i, 9, "");
                drain(s.getInputStream());            // "Unknown request 9" — 인증 없이도 응답한다
                if (i < 2) Thread.sleep(20_000);
            }
            long sec = (System.nanoTime() - t0) / 1_000_000_000L;
            System.out.println("A02 rcon unauth alive after " + sec + " s, clients=" + server.clientCount());
            // 기대(제안): 인증 전 연결은 접속 후 30초가 지나면 무조건 끊긴다.
            assertTrue(sec < RconServer.AUTH_TIMEOUT_MILLIS / 1000, "인증 없이 " + sec + "초 동안 자리를 붙잡았다(clients=" + server.clientCount() + ")");
        } finally {
            server.close();
        }
    }

    /** 루프백 주소 4개(127.0.0.2~5)에서 4개씩 = 16 자리를 채우면 관리자(127.0.0.9)는 접속하자마자 닫힌다. */
    @Test
    void fourAddressesFillAllSlots() throws Exception {
        ServerSocket ss = new ServerSocket(0, 0, InetAddress.getByName("0.0.0.0"));
        RconServer server = RconServer.open(ss, "secret", cmd -> "ran:" + cmd);
        List<Socket> held = new ArrayList<>();
        try {
            for (int ip = 2; ip <= 5; ip++) {
                for (int k = 0; k < 4; k++) {
                    Socket s = new Socket();
                    s.bind(new InetSocketAddress(InetAddress.getByName("127.0.0." + ip), 0));
                    s.connect(new InetSocketAddress(InetAddress.getLoopbackAddress(), server.port()));
                    held.add(s);
                }
            }
            long end = System.nanoTime() + 5_000_000_000L;
            while (server.clientCount() < 16 && System.nanoTime() < end) Thread.sleep(10);
            System.out.println("A02 rcon held slots = " + server.clientCount());
            Socket admin = new Socket();
            admin.bind(new InetSocketAddress(InetAddress.getByName("127.0.0.9"), 0));
            admin.connect(new InetSocketAddress(InetAddress.getLoopbackAddress(), server.port()));
            admin.setSoTimeout(3000);
            boolean ok;
            try {
                send(admin.getOutputStream(), 1, 3, "secret");
                drain(admin.getInputStream());
                ok = true;
            } catch (IOException e) {
                ok = false;
                System.out.println("A02 rcon admin refused: " + e);
            }
            admin.close();
            assertTrue(ok, "주소 4개가 RCON 자리 16개를 모두 차지해 올바른 비밀번호의 관리자가 들어오지 못했다");
        } finally {
            for (Socket s : held) s.close();
            server.close();
        }
    }

    /** 한 IP 가 원본 포트만 바꿔 핸드셰이크 1024번 → 다른 정상 클라이언트의 핸드셰이크가 응답 없음(30초 청소 전까지). */
    @Test
    void queryChallengeTableLockout() {
        AtomicLong now = new AtomicLong(1_000);
        QueryHandler.Info info = new QueryHandler.Info() {
            public String motd() { return "m"; }
            public String levelName() { return "world"; }
            public String version() { return "1.21.1"; }
            public String plugins() { return ""; }
            public int playerCount() { return 0; }
            public int maxPlayers() { return 20; }
            public int hostPort() { return 25565; }
            public String hostIp() { return "10.0.0.5"; }
            public String[] playerNames() { return new String[0]; }
        };
        QueryHandler h = new QueryHandler(info, now::get, () -> 1234);
        byte[] hs = {(byte) 0xFE, (byte) 0xFD, 9, 1, 2, 3, 4};
        for (int port = 1; port <= QueryHandler.MAX_CHALLENGES; port++) {
            assertNotNull(h.handle(new InetSocketAddress("203.0.113.7", port), hs, hs.length));
        }
        now.addAndGet(29_000);
        h.prune();
        byte[] legit = h.handle(new InetSocketAddress("198.51.100.20", 40000), hs, hs.length);
        System.out.println("A02 query legit handshake after flood = " + (legit == null ? "null" : legit.length + "B"));
        assertNotNull(legit, "한 주소의 포트 1024개로 챌린지 표가 차서 정상 질의가 막혔다");
    }
}
