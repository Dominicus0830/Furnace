// A11-02 재현: 눈 높이 칸이 단단한 블록이면 바닐라는 플레이어도 in_wall 피해 1 을 입는다(Entity.isInWall → LivingEntity.baseTick).
package org.domi.furnace.server.world;

import static org.junit.jupiter.api.Assertions.*;

import java.util.UUID;
import org.domi.furnace.api.sched.RegionPos;
import org.domi.furnace.server.ConcurrentEmbeddedChannel;
import org.domi.furnace.server.block.B;
import org.domi.furnace.server.block.Pos;
import org.domi.furnace.server.entity.Player;
import org.domi.furnace.server.net.ChunkWorkerPool;
import org.domi.furnace.server.sched.FurnaceSchedulerImpl;
import org.domi.furnace.server.world.gen.SuperflatWorldSource;
import org.junit.jupiter.api.Test;

class A11SuffocationReproTest {

    @Test
    void playerWithStoneAtEyeLevelTakesInWallDamage() throws Exception {
        FurnaceSchedulerImpl sched = new FurnaceSchedulerImpl(50_000_000L, 2);
        BlockRegistry reg = BlockRegistry.load("blocks/1.21.1.txt");
        World w = new World(new SuperflatWorldSource(reg, 0), new ChunkWorkerPool(2),
                sched, sched.regionThreadCount(), 16, 1, 1, reg.directBits(), 7, java.util.concurrent.Executors.newSingleThreadExecutor());
        Player p = new Player(1, UUID.randomUUID(), "Dom", new ConcurrentEmbeddedChannel());
        p.publishPosition(0.5, -60, 0.5, 0, 0, true);
        RegionPos rp = new RegionPos(0, 0);
        sched.region(rp).execute(() -> w.regionAt(rp).adopt(p)).get();
        for (int i = 0; i < 5; i++) sched.region(rp).execute(() -> w.regionAt(rp).tick()).get();
        sched.region(rp).execute(() -> w.regionAt(rp).level().setBlock(new Pos(0, -59, 0), B.defaultState(B.id("stone")), 3)).get();
        float before = p.vitals().health();
        for (int i = 0; i < 60; i++) sched.region(rp).execute(() -> w.regionAt(rp).tick()).get();
        assertTrue(p.vitals().health() < before, "바닐라: 10틱마다 1 피해(60틱이면 약 5). 실제 체력 " + p.vitals().health());
        sched.close();
    }
}
