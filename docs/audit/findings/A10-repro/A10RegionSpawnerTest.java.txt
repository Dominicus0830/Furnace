// A10-05 재현: 순찰 스포너가 region 마다 따로 돈다(RegionLevel.patrolSpawner 필드). 바닐라 ServerLevel 에는 PatrolSpawner(dzj) 가 하나이고
// 10~11분마다 level.players() 전체에서 한 명을 뽑는다. Furnace 는 플레이어마다 region 이 따로면 region 수만큼 뽑는다.
// 같은 시간 동안 플레이어 A 가 뽑히는 횟수를 (바닐라 모양) 스포너 하나·플레이어 둘 vs (Furnace 모양) 스포너 둘·각 한 명으로 센다.
package org.domi.furnace.server.entity;

import static org.junit.jupiter.api.Assertions.*;

import java.util.List;
import java.util.UUID;
import org.domi.furnace.server.block.BlockPlacementTest;
import org.domi.furnace.server.entity.spawn.PatrolSpawner;
import org.junit.jupiter.api.BeforeAll;
import org.junit.jupiter.api.Test;

class A10RegionSpawnerTest {

    @BeforeAll
    static void install() {
        org.domi.furnace.server.item.ItemContext.install(new org.domi.furnace.server.data.TagRegistries(
                org.domi.furnace.server.net.registry.RegistrySet.load("registries/1.21.1.txt")));
    }

    private static final class Day extends BlockPlacementTest.TestLevel {
        @Override
        protected org.domi.furnace.server.block.LevelServices services() {
            return new org.domi.furnace.server.block.LevelServices.Delegating(org.domi.furnace.server.block.LevelServices.DEFAULT) {
                @Override public long dayTime() { return 6 * 24000L + 1000L; }
                @Override public int difficulty() { return 2; }
            };
        }
    }

    private static Player player(double x) {
        var p = new Player(1, UUID.randomUUID(), "P" + x, new org.domi.furnace.server.ConcurrentEmbeddedChannel());
        p.vitals().env(new PlayerVitalsTest.TestEnv());
        p.vitals().loadGameTypes(null, org.domi.furnace.server.world.GameMode.SURVIVAL, false);
        p.applyMove(x, -60, 0.5, 0, 0, true);
        return p;
    }

    @Test
    void patrolPicksPerPlayerDoubleWithTwoRegions() {
        var level = new Day();
        ((org.domi.furnace.server.block.LegacyRandomSource) level.random()).setSeed(20261108L);
        Player a = player(0.5), b = player(5000.5);
        int[] picksA = new int[2];
        // closeToVillage 를 참으로 돌려 실제 스폰 없이 "뽑힌 플레이어"만 센다(바닐라도 이 검사 직전에 플레이어를 뽑는다)
        var vanillaShape = new PatrolSpawner(pos -> { if (pos.x() == 0) picksA[0]++; return true; });
        var regionA = new PatrolSpawner(pos -> { if (pos.x() == 0) picksA[1]++; return true; });
        var regionB = new PatrolSpawner(pos -> true);
        for (int t = 0; t < 13_200 * 600; t++) {
            vanillaShape.tick(level, List.of(a, b), true);
            regionA.tick(level, List.of(a), true);
            regionB.tick(level, List.of(b), true);
        }
        System.out.println("A 뽑힘: 스포너 하나(바닐라)=" + picksA[0] + ", region 마다(Furnace)=" + picksA[1]);
        assertTrue(picksA[1] <= picksA[0] * 1.3, "바닐라와 같다면 플레이어 A 가 뽑히는 횟수가 비슷해야 한다: 바닐라 모양 " + picksA[0] + " vs Furnace 모양 " + picksA[1]);
    }
}
