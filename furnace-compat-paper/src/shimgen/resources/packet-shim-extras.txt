        net.minecraft.world.entity.EntityType<?> type, int data, net.minecraft.world.phys.Vec3 deltaMovement, double yHeadRot) {
    this(id, uuid, type, x, y, z, velocity(deltaMovement.x), velocity(deltaMovement.y), velocity(deltaMovement.z),
            angle(xRot), angle(yRot), angle((float) yHeadRot), data);

static int velocity(double v) { return (int) (Math.max(-3.9, Math.min(3.9, v)) * 8000.0); }

static byte angle(float degrees) { return (byte) Math.floor(degrees * 256.0F / 360.0F); }

public float getXRot() { return xRot * 360 / 256.0F; }
